from pathlib import Path import re src = Path("/mnt/data/붙여넣은 텍스트 (1).txt") html = src.read_text(encoding="utf-8") # 1) JS 안에 들어간 Python식 # 주석 제거/수정 → 화면이 사라지는 원인 html = re.sub(r'^\s*#.*$', '', html, flags=re.MULTILINE) # 2) DATA_CLEAN을 만들던 블록 제거: 성조 정답 데이터가 0으로 날아가는 원인 html = re.sub( r'\nconst replacements = \{.*?\};\s*\n\s*// 퀴즈용 성조가 없는 클린 데이터 자동 생성 함수\s*\nconst DATA_CLEAN = DATA\.map\(item => \{.*?\n\}\);\s*', '\n', html, flags=re.DOTALL ) # 혹시 위 패턴이 일부만 잡힌 경우를 대비해 replacements/DATA_CLEAN 남은 블록 추가 제거 html = re.sub(r'\nconst replacements = \{.*?\};\s*', '\n', html, flags=re.DOTALL) html = re.sub(r'\nconst DATA_CLEAN = DATA\.map\(item => \{.*?\n\}\);\s*', '\n', html, flags=re.DOTALL) # 3) Q는 반드시 원본 DATA에서 생성해야 함: 정답 tone을 유지 html = html.replace( "const Q=DATA_CLEAN.map((x,i)=>({...x,id:'q'+i,lesson:lessonOf(x),tokens:x.num.split(/\\s+/).filter(Boolean).map((s,j)=>{let m=s.match(/([a-zA-ZüÜv:]+)([0-4])$/);return {raw:s,base:m?m[1]:s,tone:m?Number(m[2]):0,marked:(x.p.split(/\\s+/)[j]||'')}})}));", "const Q=DATA.map((x,i)=>({...x,id:'q'+i,lesson:lessonOf(x),tokens:x.num.split(/\\s+/).filter(Boolean).map((s,j)=>{let m=s.match(/([a-zA-ZüÜv:]+)([0-4])$/);return {raw:s,base:m?m[1]:s,tone:m?Number(m[2]):0,marked:(x.p.split(/\\s+/)[j]||'')}})}));" ) # 4) 성조퀴즈 내부 음절칸만 '알파벳 base pinyin'으로 표시 html = html.replace( '
${t.marked||t.raw}
${t.base}
', '
${t.base}
성조 선택
' ) # 5) 기존 UI 틀 유지하되 token-base가 안 사라지게, 보조문구만 작게 표시 html = html.replace( ".token-hanzi{font-size:28px;font-weight:900}.token-base{font-size:14px;color:var(--sub)}", ".token-hanzi{font-size:32px;font-weight:900;color:#143f32}.token-base{font-size:14px;color:var(--sub)}" ) out = Path("/mnt/data/dongguk_chinese_quiz_ORIGINAL_UI_FIXED.html") out.write_text(html, encoding="utf-8") print("완료:", out)
