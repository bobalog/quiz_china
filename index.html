from pathlib import Path import re src = Path("/mnt/data/dongguk_chinese_quiz_cram_green_v2.html") dst = Path("/mnt/data/dongguk_chinese_quiz_cram_green_v3.html") # 1. 원본 HTML 내용 읽어오기 html = src.read_text(encoding="utf-8") # 2. 퀴즈 전용으로 쓸 성조 없는 base 형태 딕셔너리 정의 replacements = { "méiyǒu": "mei you", "you3": "you", "nǐ": "ni", "hǎo": "hao", "shénme": "shen me", "bú": "bu", "bù": "bu", "duō": "duo", "dà": "da", "liǎng": "liang", "jǐ": "ji", "xiǎng": "xiang", "zěnme": "zen me", "wǎn": "wan", "shuì": "shui", } # 3. 원래 자바스크립트의 DATA=[...] 부분을 통째로 추출하기 data_match = re.search(r'(const DATA=\[.*?\];)', html, re.DOTALL) if data_match: original_data_block = data_match.group(1) # 4. 성조를 지운 복사본(DATA_CLEAN) 블록 만들기 clean_data_block = original_data_block # 딕셔너리 기반으로 성조 표기 단어들 교체 (예: nǐ -> ni) for old, new in replacements.items(): clean_data_block = clean_data_block.replace(old, new) # 숫자 성조(1,2,3,4) 및 알파벳 뒤에 붙은 성조 숫자 강제 제거 clean_data_block = re.sub(r'([a-zA-Z]+)[1234]', r'\1', clean_data_block) # 변수명을 DATA에서 DATA_CLEAN으로 바꿔치기 clean_data_block = clean_data_block.replace("const DATA=", "const DATA_CLEAN=") # 5. 원래 HTML 자바스크립트 로직 수정하기 # 퀴즈 로직(Q 변수 생성)은 성조가 없는 DATA_CLEAN을 쓰도록 변경! # 총망라 표는 기존처럼 성조가 살아있는 DATA를 그대로 쓰게 둠! logic_patch = 'const Q=DATA_CLEAN.map((x,i)=>' html = html.replace('const Q=DATA.map((x,i)=>', logic_patch) # 6. 생성한 DATA_CLEAN 블록을 기존 DATA 블록 바로 위에 삽입하기 final_html = html.replace(original_data_block, f"{clean_data_block}\n{original_data_block}") # 7. 변경된 내용을 새 파일에 저장! dst.write_text(final_html, encoding="utf-8") print(f"✨ 대성공! 퀴즈는 알파벳만, 총망라 표는 성조 유지 완료 -> {dst}") else: print("❌ HTML 내에서 'const DATA=[' 부분을 찾지 못했어. 파일명을 다시 확인해봐!")
